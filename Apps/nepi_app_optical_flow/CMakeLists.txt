cmake_minimum_required(VERSION 3.0.2)
project(nepi_app_optical_flow)
find_package(catkin REQUIRED COMPONENTS rospy sensor_msgs std_msgs)
catkin_package()
catkin_install_python(PROGRAMS scripts/optical_flow_app_node.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(FILES scripts/optical_flow.py scripts/radial_undistort.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
if(NOT NEPI_ENGINE)
  set(NEPI_ENGINE /opt/nepi/nepi_engine)
endif()
if(NOT NEPI_RUI)
  set(NEPI_RUI /opt/nepi/nepi_rui)
endif()
install(DIRECTORY params/ DESTINATION ${NEPI_ENGINE}/share/nepi_apps/params)
install(DIRECTORY params/ DESTINATION ${NEPI_RUI}/src/rui_webserver/rui-app/src/apps)
install(DIRECTORY rui/ DESTINATION ${NEPI_RUI}/src/rui_webserver/rui-app/src)
